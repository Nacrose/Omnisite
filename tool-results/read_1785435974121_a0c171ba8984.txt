    71→export function ProcurementModule() {
    72→  const [tab, setTab] = useState<Tab>('req')
    73→  const [selectedId, setSelectedId] = useState('REQ-0142')
    74→  const [reqs, setReqs] = useState<ReqItem[]>(() => JSON.parse(JSON.stringify(INITIAL_REQS)))
    75→  const [pos, setPos] = useState<Po[]>(() => JSON.parse(JSON.stringify(INITIAL_POS)))
    76→  // Override modal state
    77→  const [overrideModal, setOverrideModal] = useState<{ reqId: string; vendorName: string; vendorRate: number; lowestRate: number } | null>(null)
    78→  const [overrideReason, setOverrideReason] = useState('')
    79→
    80→  // Generate POs from approved requisitions — auto-group by vendor, merge duplicates
    81→  const generatePos = () => {
    82→    const approvedReqs = reqs.filter(r => r.status === 'Approved' || r.status === 'Partially PO\'d')
    83→    if (approvedReqs.length === 0) {
    84→      toast.error('No approved requisitions', { description: 'Approve requisitions first before generating POs.' })
    85→      return
    86→    }
    87→
    88→    // Group by selected vendor
    89→    const vendorGroups = new Map<string, { reqs: ReqItem[]; totalValue: number; itemCount: number }>()
    90→    for (const r of approvedReqs) {
    91→      const selectedVendor = r.vendors.find(v => v.selected)
    92→      if (!selectedVendor) continue
    93→      const existing = vendorGroups.get(selectedVendor.name) || { reqs: [], totalValue: 0, itemCount: 0 }
    94→      existing.reqs.push(r)
    95→      existing.totalValue += r.qty * selectedVendor.rate
    96→      existing.itemCount += 1
    97→      vendorGroups.set(selectedVendor.name, existing)
    98→    }
    99→
   100→    // Create one PO per vendor
   101→    const newPOs: Po[] = []
   102→    let poNum = 19 // starting after existing PO-018
   103→    for (const [vendor, group] of vendorGroups) {
   104→      const po: Po = {
   105→        id: `PO-2410-${String(poNum).padStart(3, '0')}`,
   106→        vendor,
   107→        date: new Date().toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' }),
   108→        value: group.totalValue,
   109→        status: 'Pending',
   110→        items: group.itemCount,
   111→        grn: false,
   112→      }
   113→      newPOs.push(po)
   114→      poNum++
   115→    }
   116→
   117→    setPos(prev => [...newPOs, ...prev])
   118→
   119→    // Mark requisitions as Fully PO'd
   120→    setReqs(prev => prev.map(r => {
   121→      if (approvedReqs.find(ar => ar.id === r.id)) {
   122→        return { ...r, status: 'Fully PO\'d' as const }
   123→      }
   124→      return r
   125→    }))
   126→
   127→    toast.success(`${newPOs.length} PO${newPOs.length > 1 ? 's' : ''} generated`, {
   128→      description: newPOs.map(p => `${p.id} → ${p.vendor} · NPR ${p.value.toLocaleString()}`).join('\n'),
   129→    })
   130→
   131→    // Switch to PO tab to show the new POs
   132→    setTab('po')
   133→  }
   134→
   135→  // Select a vendor. If the vendor is NOT the lowest bidder, open the override modal.
   136→  const selectVendor = (reqId: string, vendorName: string) => {
   137→    const req = reqs.find(r => r.id === reqId)
   138→    if (!req) return
   139→    const vendor = req.vendors.find(v => v.name === vendorName)
   140→    if (!vendor) return
   141→    const lowest = Math.min(...req.vendors.map(v => v.rate))
   142→    if (vendor.rate > lowest && !vendor.selected) {
   143→      // Open override modal — don't select yet
   144→      setOverrideModal({ reqId, vendorName, vendorRate: vendor.rate, lowestRate: lowest })
   145→      setOverrideReason('')
   146→    } else {
   147→      // Select directly (lowest bidder or re-selecting already-selected)
   148→      applyVendorSelection(reqId, vendorName)
   149→      toast.success(`${vendorName} selected`, {
   150→        description: vendor.rate === lowest ? 'Lowest bidder auto-confirmed' : 'Vendor selected',
   151→      })
   152→    }
   153→  }
   154→
   155→  const applyVendorSelection = (reqId: string, vendorName: string) => {
   156→    setReqs(prev => prev.map(r => r.id === reqId ? {
   157→      ...r,
   158→      vendors: r.vendors.map(v => ({ ...v, selected: v.name === vendorName })),
   159→    } : r))
   160→  }
   161→
   162→  const confirmOverride = () => {
   163→    if (!overrideModal) return
   164→    if (!overrideReason.trim()) {
   165→      toast.error('Justification required', { description: 'Please provide a reason for overriding the lowest bidder.' })
   166→      return
   167→    }
   168→    applyVendorSelection(overrideModal.reqId, overrideModal.vendorName)
   169→    // Save the reason on the req
   170→    setReqs(prev => prev.map(r => r.id === overrideModal.reqId ? { ...r, overrideReason: overrideReason.trim() } : r))
   171→    toast.success('Override approved', {
   172→      description: `${overrideModal.vendorName} selected with justification. Audit logged.`,
   173→    })
   174→    setOverrideModal(null)
   175→    setOverrideReason('')
   176→  }
   177→
   178→  return (
   179→    <>
   180→      <Toaster richColors position="top-center" />
   181→      <Workspace2Pane
   182→        leftPane={
   183→          <>
   184→            <PaneHeader title="Procurement">
   185→              <Button variant="ghost" size="sm" className="h-7"><Plus className="w-3.5 h-3.5" /></Button>
   186→            </PaneHeader>
   187→            <div className="py-2">
   188→              {([
   189→                { id: 'req', name: 'Requisitions', count: reqs.length, icon: FileText },
   190→                { id: 'po', name: 'Purchase Orders', count: pos.length, icon: Package },
   191→                { id: 'grn', name: 'GRN / 3-Way Match', count: 4, icon: CheckCircle2 },
   192→                { id: 'stock', name: 'Live Stock', count: 5, icon: Boxes },
   193→                { id: 'min', name: 'Material Issues (MIN)', count: 8, icon: ArrowRight },
   194→              ] as { id: Tab; name: string; count: number; icon: typeof FileText }[]).map(t => {
   195→                const Icon = t.icon
   196→                return (
   197→                  <button
   198→                    key={t.id}
   199→                    onClick={() => setTab(t.id)}
   200→                    className={cn(
   201→                      'w-full flex items-center gap-2.5 h-9 px-4 text-xs transition-colors',
   202→                      tab === t.id ? 'bg-accent border-l-2 border-primary' : 'hover:bg-accent/50 border-l-2 border-transparent'
   203→                    )}
   204→                  >
   205→                    <Icon className="w-3.5 h-3.5 text-muted-foreground" />
   206→                    <span className="flex-1 text-left">{t.name}</span>
   207→                    <Badge variant="secondary" className="text-[9px] h-4 px-1">{t.count}</Badge>
   208→                  </button>
   209→                )
   210→              })}
   211→            </div>
   212→            <div className="mt-auto border-t border-[var(--pane-divider)] p-3 space-y-1 text-xs">
   213→              <div className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">Procurement Snapshot</div>
   214→              <div className="flex justify-between"><span className="text-muted-foreground">Open POs</span><span className="font-mono">12</span></div>
   215→              <div className="flex justify-between"><span className="text-muted-foreground">Committed cost</span><span className="font-mono">NPR 18.4M</span></div>
   216→              <div className="flex justify-between"><span className="text-muted-foreground">Stock value</span><span className="font-mono">NPR 1.92M</span></div>
   217→            </div>
   218→          </>
   219→        }
   220→        centerPane={
   221→          <>
   222→            <PaneHeader title={tab === 'req' ? 'Requisitions & Comparative Statement' : tab === 'po' ? 'Purchase Orders' : tab === 'grn' ? 'GRN & 3-Way Match' : tab === 'stock' ? 'Live Stock Dashboard' : 'Material Issue Notes'}>
   223→              <Button variant="ghost" size="sm" className="h-7 text-xs gap-1.5"><Search className="w-3.5 h-3.5" />Search</Button>
   224→              <Button size="sm" className="h-7 text-xs gap-1.5" onClick={() => tab === 'grn' ? toast.info('GRN Receiving Form', { description: 'Select a PO, enter received qty, attach delivery note. System verifies against PO qty.' }) : undefined}><Plus className="w-3.5 h-3.5" />New {tab === 'req' ? 'Requisition' : tab === 'po' ? 'Consolidated PO' : tab === 'grn' ? 'GRN' : tab === 'stock' ? 'Material' : 'MIN'}</Button>
   225→            </PaneHeader>
   226→
   227→            {tab === 'req' && (
   228→              <ReqCenterView reqs={reqs} selectedId={selectedId} onSelect={setSelectedId} onVendorSelect={selectVendor} onGeneratePos={generatePos} />
   229→            )}
   230→            {tab === 'po' && <PoCenterView pos={pos} />}
   231→            {tab === 'grn' && <GrnCenterView />}
   232→            {tab === 'stock' && <StockCenterView />}
   233→            {tab === 'min' && <MinCenterView />}
   234→          </>
   235→        }

   237→        leftPaneWidth="260px"
   238→        rightPaneWidth="380px"
   239→      />
   240→
   241→      {/* Override Justification Modal */}
   242→      {overrideModal && (
   243→        <div
   244→          className="fixed inset-0 z-50 bg-black/40 backdrop-blur-sm flex items-center justify-center p-4"
   245→          onClick={() => setOverrideModal(null)}
   246→        >
   247→          <div
   248→            className="w-full max-w-md pane border border-[var(--pane-divider)] rounded-xl shadow-2xl overflow-hidden"
   249→            onClick={e => e.stopPropagation()}
   250→          >
   251→            <div className="flex items-center justify-between px-4 h-12 border-b border-[var(--pane-divider)] bg-amber-500/10">
   252→              <div className="flex items-center gap-2">
   253→                <ShieldAlert className="w-4 h-4 text-amber-500" />
   254→                <span className="text-sm font-semibold">Override Justification Required</span>
   255→              </div>
   256→              <button onClick={() => setOverrideModal(null)} className="p-1 rounded hover:bg-accent text-muted-foreground">
   257→                <X className="w-4 h-4" />
   258→              </button>
   259→            </div>
   260→            <div className="p-4 space-y-3">
   261→              <div className="text-xs text-muted-foreground">
   262→                You are selecting a vendor that is <span className="font-semibold text-amber-600">NPR {(overrideModal.vendorRate - overrideModal.lowestRate).toLocaleString()}</span> above the lowest bidder. FIDIC Clause 4.1 requires documented justification for non-lowest bids.
   263→              </div>
   264→              <div className="grid grid-cols-2 gap-3 text-xs">

   291→              </div>
   292→            </div>
   293→          </div>
   294→        </div>
   295→      )}
   296→    </>

   300→function ReqCenterView({ reqs, selectedId, onSelect, onVendorSelect, onGeneratePos }: {
   301→  reqs: ReqItem[]; selectedId: string; onSelect: (id: string) => void; onVendorSelect: (reqId: string, vendorName: string) => void; onGeneratePos: () => void
   302→}) {
   303→  return (
   304→    <>
   305→      <div className="px-4 py-3 border-b border-[var(--pane-divider)] bg-secondary/20 text-xs text-muted-foreground">
   306→        Selecting lowest bidder is automatic (🏆). Choosing a higher bidder requires justification. Click a vendor card to select.
   307→      </div>
   308→      <div className="space-y-3 p-3">
   309→        {reqs.map(r => {
   310→          const lowest = Math.min(...r.vendors.map(v => v.rate))
   311→          const selectedVendor = r.vendors.find(v => v.selected)
   312→          const isOverride = selectedVendor && selectedVendor.rate > lowest
   313→          return (
   314→            <div
   315→              key={r.id}
   316→              onClick={() => onSelect(r.id)}
   317→              className={cn(
   318→                'rounded-lg border p-3 cursor-pointer transition-colors',
   319→                selectedId === r.id ? 'border-primary bg-accent/40' : 'border-[var(--pane-divider)] hover:border-primary/40'
   320→              )}
   321→            >
   322→              <div className="flex items-center gap-2 mb-2">
   323→                <span className="font-mono text-xs text-muted-foreground">{r.id}</span>
   324→                <Badge variant="outline" className="text-[10px]">{r.source}</Badge>
   325→                <Badge variant="secondary" className={cn('text-[10px]', r.status === 'Fully PO\'d' && 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-300', r.status === 'Partially PO\'d' && 'bg-amber-500/15 text-amber-700 dark:text-amber-300')}>{r.status}</Badge>
   326→                <span className="ml-auto text-xs text-muted-foreground">{r.qty} {r.uom}</span>
   327→              </div>
   328→              <div className="font-medium text-sm">{r.item}</div>
   329→              {/* Vendor matrix — now interactive */}
   330→              <div className="mt-2 grid grid-cols-3 gap-2">
   331→                {r.vendors.map((v, i) => {
   332→                  const isLowest = v.rate === lowest
   333→                  return (
   334→                    <button
   335→                      key={i}
   336→                      onClick={(e) => { e.stopPropagation(); onVendorSelect(r.id, v.name) }}
   337→                      className={cn(
   338→                        'p-2 rounded border text-xs text-left transition-all hover:shadow-sm',
   339→                        v.selected ? 'border-primary bg-primary/5 ring-1 ring-primary/30' : 'border-[var(--pane-divider)] hover:border-primary/40 hover:bg-accent/30',
   340→                        isLowest && !v.selected && 'border-emerald-500/40'
   341→                      )}
   342→                    >
   343→                      <div className="flex items-center justify-between">
   344→                        <span className="font-medium truncate">{v.name}</span>
   345→                        {isLowest && <Trophy className="w-3 h-3 text-amber-500 flex-shrink-0" />}
   346→                      </div>
   347→                      <div className="font-mono mt-0.5">NPR {v.rate.toLocaleString()}</div>
   348→                      {v.selected && <div className="text-[9px] text-primary mt-0.5 font-semibold">✓ Selected</div>}
   349→                      {isLowest && !v.selected && <div className="text-[9px] text-emerald-600 mt-0.5">Lowest bid</div>}
   350→                    </button>
   351→                  )
   352→                })}
   353→              </div>
   354→              {isOverride && (
   355→                <div className="mt-2 p-2 rounded bg-amber-500/10 border border-amber-500/30 text-[11px] flex items-start gap-1.5">
   356→                  <AlertTriangle className="w-3 h-3 text-amber-500 mt-0.5 flex-shrink-0" />
   357→                  <div>
   358→                    <span className="font-medium">Override justification on file:</span>
   359→                    <span className="text-muted-foreground"> NPR {(selectedVendor!.rate - lowest).toLocaleString()} above lowest. "{r.overrideReason || 'Better delivery lead-time (3 days vs 7 days)'}"</span>
   360→                  </div>
   361→                </div>
   362→              )}
   363→            </div>
   364→          )
   365→        })}
   366→      </div>
   367→
   368→      <div className="border-t border-[var(--pane-divider)] p-3 bg-secondary/20">
   369→        <div className="flex items-center justify-between mb-2">
   370→          <span className="text-xs font-semibold">Consolidated PO Builder</span>
   371→          <Button size="sm" className="h-7 text-xs gap-1.5" onClick={onGeneratePos}><Package className="w-3.5 h-3.5" />Generate POs</Button>
   372→        </div>
   373→        <div className="text-[11px] text-muted-foreground">
   374→          {reqs.filter(r => r.status === 'Approved' || r.status === 'Partially PO\'d').length} approved requisitions will be auto-grouped by vendor and merged into POs. Pushes "Committed Cost" to Financials.
   375→        </div>
   376→      </div>
   377→    </>
   378→  )
