    21→// ─── Types ───────────────────────────────────────────────────────────────────
    22→
    23→type ItemType = 'composite' | 'conditional'
    24→
    25→interface ScItem {
    26→  id: string
    27→  code: string
    28→  desc: string
    29→  uom: string
    30→  rate: number
    31→  plannedQty: number
    32→  actualQty: number
    33→  type: ItemType
    34→  // Mapping to main BOQ (for composite items — e.g., "drain per rmt" maps to excavation, PCC, RCC, etc.)
    35→  mapping?: { boqCode: string; boqDesc: string; coefficient: number; uom: string }[]
    36→  // For conditional items (tunneling support): rock class + design pattern
    37→  rockClass?: string
    38→  designPattern?: number  // expected qty per rm of advance for this rock class
    39→}
    40→
    41→interface MaterialIssue {
    42→  id: string       // MIN number
    43→  date: string
    44→  materialCode: string
    45→  materialName: string
    46→  uom: string
    47→  qty: number
    48→  rate: number
    49→  issuedBy: string
    50→  notes?: string
    51→}
    52→
    53→interface MaterialReturn {
    54→  id: string       // MRN number
    55→  date: string
    56→  materialCode: string
    57→  materialName: string
    58→  uom: string
    59→  qty: number
    60→  rate: number
    61→  returnedBy: string
    62→  notes?: string
    63→}
    64→
    65→interface ConsumableIssue {
    66→  id: string
    67→  date: string
    68→  name: string       // curing compound, binding wire, diesel, form release agent
    69→  uom: string
    70→  qty: number
    71→  rate: number
    72→  normPerUnit?: number  // e.g., 0.5 kg binding wire per MT steel
    73→  normUnit?: string     // "MT"
    74→  normBasis?: number    // total basis (e.g., 28.5 MT steel)
    75→}
    76→
    77→interface CustomDeductible {
    78→  id: string
    79→  type: 'tds' | 'equipment' | 'penalty' | 'electricity' | 'insurance' | 'material_overuse' | 'other'
    80→  label: string
    81→  amount: number
    82→  ratePct?: number
    83→  notes?: string
    84→}

    88→  name: string
    89→  scope: string

    92→  advancePct: number       // e.g., 10%

   106→  ncrCount: number
   107→  incidents: number
   108→  isTunneling: boolean
   109→}

   325→
   326→function fmt(n: number) {
   327→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`
   328→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   329→  return n.toFixed(0)
   330→}
   331→
   332→function fmtNPR(n: number) {
   333→  return `NPR ${n.toLocaleString('en-IN')}`
   334→}

   433→// ─── Sub-BOQ Tab (composite + conditional items + mapping) ───────────────────
   434→
   435→function SubBoqTab({ sc }: { sc: Subcontractor }) {
   436→  const compositeItems = sc.items.filter(i => i.type === 'composite')
   437→  const conditionalItems = sc.items.filter(i => i.type === 'conditional')
   438→  const earned = sc.items.reduce((sum, it) => sum + it.actualQty * it.rate, 0)
   439→
   440→  return (
   441→    <div className="p-4 space-y-4 text-xs">
   442→      {/* Earned value summary */}
   443→      <div className="p-2.5 rounded-md bg-secondary/40">
   444→        <div className="text-[10px] text-muted-foreground">Total Earned Value (SC BOQ actuals × SC rates)</div>
   445→        <div className="text-lg font-bold mt-0.5 tabular-nums">{fmtNPR(earned)}</div>
   446→      </div>
   447→
   448→      {/* Composite items */}
   449→      <div>
   450→        <div className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground mb-2 flex items-center gap-1.5">
   451→          <Layers className="w-3 h-3" /> Composite Items ({compositeItems.length})
   452→        </div>
   453→        <div className="space-y-2">
   454→          {compositeItems.map(it => {
   455→            const earnedItem = it.actualQty * it.rate
   456→            const progress = it.plannedQty > 0 ? (it.actualQty / it.plannedQty) * 100 : 0
   457→            return (

   798→// ─── Running Bill Tab (expanded deductibles) ─────────────────────────────────

   800→function RunningBillTab({ sc }: { sc: Subcontractor }) {
   801→  const earned = sc.items.reduce((sum, it) => sum + it.actualQty * it.rate, 0)
   802→  const retention = earned * (sc.retentionPct / 100)
   803→  const tds = sc.customDeductibles.find(d => d.type === 'tds')
   804→  const tdsAmount = tds ? earned * ((tds.ratePct || 0) / 100) : 0
   805→  const otherDeductibles = sc.customDeductibles.filter(d => d.type !== 'tds')
   806→  const otherDeductibleTotal = otherDeductibles.reduce((sum, d) => sum + d.amount, 0)
   807→
   808→  // Material over-use chargeback
   809→  let materialChargeback = 0
   810→  const materialMap = new Map<string, { issued: number; returned: number; theoretical: number; rate: number }>()
   811→  for (const mi of sc.materialIssues) {
   812→    const e = materialMap.get(mi.materialCode) || { issued: 0, returned: 0, theoretical: 0, rate: mi.rate }
   813→    e.issued += mi.qty; materialMap.set(mi.materialCode, e)
   814→  }
   815→  for (const mr of sc.materialReturns) {
   816→    const e = materialMap.get(mr.materialCode)
   817→    if (e) e.returned += mr.qty
   818→  }
   819→  const totalRmt = sc.items.find(i => i.type === 'composite')?.actualQty || 0
   820→  for (const [, m] of materialMap) {
   821→    if (m.code === 'M-CEM-OPC' as any) m.theoretical = totalRmt * 5.7
   822→    else if (m.code === 'M-STEEL-TMT16' as any || m.code === 'M-STEEL-ISMB150' as any) m.theoretical = totalRmt * 0.095
   823→    const netUsed = m.issued - m.returned
   824→    const overQty = Math.max(0, netUsed - m.theoretical)
   825→    materialChargeback += overQty * m.rate
   826→  }
   827→
   828→  // Consumable chargeback
   829→  let consumableChargeback = 0
   830→  sc.consumables.forEach(c => {
   831→    const theoretical = c.normPerUnit && c.normBasis ? c.normPerUnit * c.normBasis : 0
   832→    const overQty = Math.max(0, c.qty - theoretical)
   833→    consumableChargeback += overQty * c.rate
   834→  })
   835→
   836→  const totalDeductions = sc.advancePaid + retention + sc.reworkCost + tdsAmount + otherDeductibleTotal + materialChargeback + consumableChargeback
   837→  const netPayable = earned - totalDeductions

   989→function PerformanceTab({ sc }: { sc: Subcontractor }) {
   990→  const onTimeRate = sc.assignedTasks.length > 0
   991→    ? (sc.assignedTasks.filter(t => t.status === 'on-track').length / sc.assignedTasks.length) * 100
   992→    : 100
   993→
   994→  const earned = sc.items.reduce((sum, it) => sum + it.actualQty * it.rate, 0)
   995→  const retention = earned * (sc.retentionPct / 100)
   996→  const netPayable = earned - sc.advancePaid - retention - sc.reworkCost
   997→
   998→  // Material efficiency
   999→  let matEfficiency = 100
  1000→  let matCount = 0
  1001→  const materialMap = new Map<string, { issued: number; returned: number; theoretical: number }>()
  1002→  for (const mi of sc.materialIssues) {
  1003→    const e = materialMap.get(mi.materialCode) || { issued: 0, returned: 0, theoretical: 0 }
  1004→    e.issued += mi.qty; materialMap.set(mi.materialCode, e)
  1005→  }
  1006→  for (const mr of sc.materialReturns) {
  1007→    const e = materialMap.get(mr.materialCode)
  1008→    if (e) e.returned += mr.qty
  1009→  }
  1010→  const totalRmt = sc.items.find(i => i.type === 'composite')?.actualQty || 0
  1011→  for (const [, m] of materialMap) {
  1012→    if (m.code === 'M-CEM-OPC' as any) m.theoretical = totalRmt * 5.7
  1013→    else if (m.code === 'M-STEEL-TMT16' as any || m.code === 'M-STEEL-ISMB150' as any) m.theoretical = totalRmt * 0.095
  1014→    const netUsed = m.issued - m.returned
  1015→    const variance = m.theoretical > 0 ? Math.abs(((netUsed - m.theoretical) / m.theoretical) * 100) : 0
  1016→    matEfficiency = Math.min(matEfficiency, 100 - variance)
  1017→    matCount++
  1018→  }
  1019→
  1020→  const kpis = [
  1021→    { label: 'On-Time Delivery', value: `${onTimeRate.toFixed(0)}%`, icon: Calendar, color: onTimeRate >= 80 ? 'text-emerald-600' : onTimeRate >= 50 ? 'text-amber-600' : 'text-red-600', desc: `${sc.assignedTasks.filter(t => t.status === 'on-track').length}/${sc.assignedTasks.length} tasks on track` },
  1022→    { label: 'Quality (NCRs)', value: `${sc.ncrCount}`, icon: ShieldCheck, color: sc.ncrCount === 0 ? 'text-emerald-600' : sc.ncrCount <= 1 ? 'text-amber-600' : 'text-red-600', desc: 'Non-conformance reports linked to SC' },
  1023→    { label: 'Material Efficiency', value: `${matEfficiency.toFixed(0)}%`, icon: Package, color: matEfficiency >= 95 ? 'text-emerald-600' : matEfficiency >= 85 ? 'text-amber-600' : 'text-red-600', desc: `${matCount} materials tracked` },
  1024→    { label: 'Safety (Incidents)', value: `${sc.incidents}`, icon: Activity, color: sc.incidents === 0 ? 'text-emerald-600' : 'text-red-600', desc: 'Incidents on SC tasks' },
  1025→  ]
  1026→
  1027→  return (
  1028→    <div className="p-4 space-y-3 text-xs">
  1029→      <div className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">Performance Dashboard</div>
  1030→
  1031→      {/* KPI grid */}
  1032→      <div className="grid grid-cols-2 gap-2">
  1033→        {kpis.map(k => {
  1034→          const Icon = k.icon
  1035→          return (
  1036→            <div key={k.label} className="p-2.5 rounded-md border border-[var(--pane-divider)]">
  1037→              <div className="flex items-center justify-between">
  1038→                <span className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground">{k.label}</span>
  1039→                <Icon className={cn('w-3.5 h-3.5', k.color)} />
  1040→              </div>
  1041→              <div className={cn('text-lg font-bold mt-0.5', k.color)}>{k.value}</div>
  1042→              <div className="text-[10px] text-muted-foreground">{k.desc}</div>
  1043→            </div>
  1044→          )
  1045→        })}
  1046→      </div>
  1047→
  1048→      {/* Compliance */}
  1049→      <div>
  1050→        <div className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground mb-2">Compliance</div>
  1051→        <div className="space-y-1.5">
  1052→          <ComplianceRow label="PAN" value={sc.pan} status="ok" />
  1053→          <ComplianceRow label="GST" value={sc.gst} status="ok" />
  1054→          <ComplianceRow label="Insurance" value={`Expires ${sc.insuranceExpiry}`} status="ok" />
  1055→          <ComplianceRow label="Labour License" value={`Expires ${sc.labourLicenseExpiry}`} status="warn" />
  1056→        </div>
  1057→      </div>
  1058→
  1059→      {/* Financial summary */}
  1060→      <div>
  1061→        <div className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground mb-2">Financial Summary</div>
  1062→        <div className="p-2.5 rounded-md bg-secondary/40 space-y-1">
  1063→          <div className="flex justify-between"><span className="text-muted-foreground">Agreement value</span><span className="font-mono">{sc.agreementValue > 0 ? fmtNPR(sc.agreementValue) : 'Variable'}</span></div>
  1064→          <div className="flex justify-between"><span className="text-muted-foreground">Earned to date</span><span className="font-mono font-medium">{fmtNPR(earned)}</span></div>
  1065→          <div className="flex justify-between"><span className="text-muted-foreground">Advance paid</span><span className="font-mono text-red-600">{fmtNPR(sc.advancePaid)}</span></div>
  1066→          <div className="flex justify-between"><span className="text-muted-foreground">Net payable</span><span className="font-mono font-bold text-emerald-600">{fmtNPR(netPayable)}</span></div>
  1067→        </div>
  1068→      </div>
  1069→    </div>
  1070→  )
