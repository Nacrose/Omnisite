    79→    79→function flattenTasks(items: Task[]): { task: Task; depth: number }[] {
    80→    80→  const out: { task: Task; depth: number }[] = []
    81→    81→  const walk = (items: Task[], depth: number) => {
    82→    82→    for (const t of items) {
    83→    83→      out.push({ task: t, depth })
    84→    84→      if (t.children) walk(t.children, depth + 1)
    85→    85→    }
    86→    86→  }
    87→    87→  walk(items, 0)
    88→    88→  return out
    89→    89→}
