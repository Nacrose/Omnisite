   326→   326→function fmt(n: number) {
   327→   327→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`
   328→   328→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`
   329→   329→  return n.toFixed(0)
   330→   330→}
   331→   331→
   332→   332→function fmtNPR(n: number) {
   333→   333→  return `NPR ${n.toLocaleString('en-IN')}`
   334→   334→}

   581→   581→function MaterialTab({ sc }: { sc: Subcontractor }) {
   582→   582→  // Aggregate by material
   583→   583→  const materialMap = new Map<string, {
   584→   584→    code: string; name: string; uom: string; rate: number;
   585→   585→    issued: number; returned: number; theoretical: number;
   586→   586→  }>()
   587→   587→
   588→   588→  // Sum issued
   589→   589→  for (const mi of sc.materialIssues) {
   590→   590→    const key = mi.materialCode
   591→   591→    const existing = materialMap.get(key) || { code: mi.materialCode, name: mi.materialName, uom: mi.uom, rate: mi.rate, issued: 0, returned: 0, theoretical: 0 }
   592→   592→    existing.issued += mi.qty
   593→   593→    materialMap.set(key, existing)
   594→   594→  }
   595→   595→  // Sum returns
   596→   596→  for (const mr of sc.materialReturns) {
   597→   597→    const key = mr.materialCode
   598→   598→    const existing = materialMap.get(key)
   599→   599→    if (existing) existing.returned += mr.qty
   600→   600→  }
   601→   601→  // Calculate theoretical from composite items mapping × RA coefficients
   602→   602→  // Simplified: cement = 4.5 bags/cum PCC, steel = 1:1, aggregate = 0.9 cum/cum, sand = 0.45 cum/cum
   603→   603→  const totalRmt = sc.items.find(i => i.type === 'composite')?.actualQty || 0
   604→   604→  for (const [, m] of materialMap) {
   605→   605→    if (m.code === 'M-CEM-OPC') {
   606→   606→      // PCC: 0.40 cum/rmt × 4.5 bags + RCC: 0.60 cum/rmt × 6.5 bags = 5.7 bags/rmt
   607→   607→      m.theoretical = totalRmt * 5.7
   608→   608→    } else if (m.code === 'M-STEEL-TMT16' || m.code === 'M-STEEL-ISMB150') {
   609→   609→      m.theoretical = totalRmt * 0.095
   610→   610→    } else if (m.code === 'M-AGG-20') {
   611→   611→      m.theoretical = totalRmt * (0.40 * 0.9 + 0.60 * 0.9) // PCC + RCC agg
   612→   612→    } else if (m.code === 'M-SAND-R') {
   613→   613→      m.theoretical = totalRmt * (0.40 * 0.45 + 0.60 * 0.45)
   614→   614→    }
   615→   615→  }
   616→   616→
   617→   617→  const materials = Array.from(materialMap.values())
   618→   618→
   619→   619→  return (
   620→   620→    <div className="p-4 space-y-3 text-xs">
   621→   621→      <div className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   622→   622→        Material Issue & Reconciliation
   623→   623→      </div>
   624→   624→
   625→   625→      {/* Summary cards */}
   626→   626→      <div className="grid grid-cols-3 gap-2">
   627→   627→        <div className="p-2 rounded-md bg-secondary/40 text-center">
   628→   628→          <div className="text-[10px] text-muted-foreground">Total Issued</div>
   629→   629→          <div className="text-sm font-bold">{sc.materialIssues.length} MINs</div>
   630→   630→        </div>
   631→   631→        <div className="p-2 rounded-md bg-secondary/40 text-center">
   632→   632→          <div className="text-[10px] text-muted-foreground">Total Returns</div>
   633→   633→          <div className="text-sm font-bold">{sc.materialReturns.length} MRNs</div>
   634→   634→        </div>
   635→   635→        <div className="p-2 rounded-md bg-secondary/40 text-center">
   636→   636→          <div className="text-[10px] text-muted-foreground">Materials Tracked</div>
   637→   637→          <div className="text-sm font-bold">{materials.length}</div>
   638→   638→        </div>
   639→   639→      </div>
   640→   640→
   641→   641→      {/* Reconciliation table */}
   642→   642→      <div className="rounded-md border border-[var(--pane-divider)] overflow-hidden">
   643→   643→        <div className="grid grid-cols-12 gap-1 px-2 py-1.5 bg-secondary/30 text-[9px] font-semibold uppercase tracking-wider text-muted-foreground">
   644→   644→          <div className="col-span-4">Material</div>
   645→   645→          <div className="col-span-2 text-right">Theoretical</div>
   646→   646→          <div className="col-span-2 text-right">Issued</div>
   647→   647→          <div className="col-span-1 text-right">Ret.</div>
   648→   648→          <div className="col-span-2 text-right">Net Used</div>
   649→   649→          <div className="col-span-1 text-right">Var%</div>
   650→   650→        </div>

   659→   659→                <div className="text-[9px] text-muted-foreground font-mono">{m.code}</div>

   800→   800→function RunningBillTab({ sc }: { sc: Subcontractor }) {

   803→   803→  const tds = sc.customDeductibles.find(d => d.type === 'tds')

   806→   806→  const otherDeductibleTotal = otherDeductibles.reduce((sum, d) => sum + d.amount, 0)
   807→   807→

   810→   810→  const materialMap = new Map<string, { issued: number; returned: number; theoretical: number; rate: number }>()
   811→   811→  for (const mi of sc.materialIssues) {
   812→   812→    const e = materialMap.get(mi.materialCode) || { issued: 0, returned: 0, theoretical: 0, rate: mi.rate }
   813→   813→    e.issued += mi.qty; materialMap.set(mi.materialCode, e)
   814→   814→  }
   815→   815→  for (const mr of sc.materialReturns) {
   816→   816→    const e = materialMap.get(mr.materialCode)
   817→   817→    if (e) e.returned += mr.qty
   818→   818→  }
   819→   819→  const totalRmt = sc.items.find(i => i.type === 'composite')?.actualQty || 0
   820→   820→  for (const [, m] of materialMap) {
   821→   821→    if (m.code === 'M-CEM-OPC' as any) m.theoretical = totalRmt * 5.7
   822→   822→    else if (m.code === 'M-STEEL-TMT16' as any || m.code === 'M-STEEL-ISMB150' as any) m.theoretical = totalRmt * 0.095
   823→   823→    const netUsed = m.issued - m.returned
   824→   824→    const overQty = Math.max(0, netUsed - m.theoretical)
   825→   825→    materialChargeback += overQty * m.rate
   826→   826→  }
   827→   827→
   828→   828→  // Consumable chargeback
   829→   829→  let consumableChargeback = 0
   830→   830→  sc.consumables.forEach(c => {
   831→   831→    const theoretical = c.normPerUnit && c.normBasis ? c.normPerUnit * c.normBasis : 0
   832→   832→    const overQty = Math.max(0, c.qty - theoretical)
   833→   833→    consumableChargeback += overQty * c.rate
   834→   834→  })
   835→   835→
   836→   836→  const totalDeductions = sc.advancePaid + retention + sc.reworkCost + tdsAmount + otherDeductibleTotal + materialChargeback + consumableChargeback

   849→   849→    other: FileText,
