    94→function flatten(items: BoqItem[]): BoqItem[] {
    95→  const out: BoqItem[] = []
    96→  for (const i of items) {
    97→    out.push(i)
    98→    if (i.children) out.push(...flatten(i.children))
    99→  }
   100→  return out
   101→}

   337→  // Find an item and its parent in the tree
   338→  const findItemAndParent = (items: BoqItem[], id: string, parent: BoqItem | null = null, depth = 0): { item: BoqItem; parent: BoqItem | null; depth: number } | null => {
   339→    for (const it of items) {
   340→      if (it.id === id) return { item: it, parent, depth }
   341→      if (it.children) {
   342→        const found = findItemAndParent(it.children, id, it, depth + 1)
   343→        if (found) return found
   344→      }
   345→    }
   346→    return null

   353→      level: newLevel,
   354→      children: item.children?.map(c => updateLevels(c, newLevel + 1)),

   358→  // Reparent an item: remove from old location, add to new parent's children
   359→  const reparentItem = (draggedId: string, targetHeadingId: string) => {
   360→    if (draggedId === targetHeadingId) return // can't drop on self
   361→
   362→    // Find the dragged item and check it's not a parent of the target (no cycles)
   363→    const dragInfo = findItemAndParent(boqData, draggedId)
   364→    if (!draggedItem) return
   365→
   366→    // Check for cycle: is targetHeadingId a descendant of draggedId?
   367→    const isDescendant = (items: BoqItem[] | undefined, ancestorId: string, targetId: string): boolean => {
   368→      if (!items) return false
   369→      for (const it of items) {
   370→        if (it.id === ancestorId) {
   371→          // Check if targetId is in this subtree
   372→          const checkSubtree = (node: BoqItem): boolean => {
   373→            if (node.id === targetId) return true
   374→            return node.children?.some(c => checkSubtree(c)) || false
   375→          }
   376→          if (checkSubtree(it)) return true
   377→        }
   378→        if (it.children && isDescendant(it.children, ancestorId, targetId)) return true
   379→      }
   380→      return false
   381→    }
   382→    if (isDescendant(boqData, draggedId, targetHeadingId)) {
   383→      toast.error('Cannot reparent', { description: 'Cannot move a heading into its own subtree' })
   384→      return
   385→    }
   386→
   387→    commitBoqData(prev => {
   388→      const updated = JSON.parse(JSON.stringify(prev)) as BoqItem[]
   389→      let movedItem: BoqItem | null = null
   390→
   391→      // Step 1: Remove the dragged item from its current location
   392→      const removeFromTree = (items: BoqItem[]): BoqItem[] => {
   393→        return items.filter(it => {
   394→          if (it.id === draggedId) {
   395→            movedItem = it
   396→            return false
   397→          }
   398→          if (it.children) it.children = removeFromTree(it.children)
   399→          return true
   400→        })
   401→      }
   402→      const cleaned = removeFromTree(updated)
   403→
   404→      // Step 2: Find the target heading and add the item to its children
   405→      if (movedItem) {
   406→        const targetLevel = findItemAndParent(cleaned, targetHeadingId)?.depth ?? 0
   407→        movedItem = updateLevels(movedItem, targetLevel + 1)
   408→        const addToTarget = (items: BoqItem[]): boolean => {
   409→          for (const it of items) {
   410→            if (it.id === targetHeadingId) {
   411→              if (!it.children) it.children = []
   412→              it.children.push(movedItem!)
   413→              return true
   414→            }
   415→            if (it.children && addToTarget(it.children)) return true
   416→          }
   417→          return false
   418→        }
   419→        addToTarget(cleaned)
   420→      }
   421→
   422→      return cleaned
   423→    })
   424→

   426→    setExpandedArr(prev => prev.includes(targetHeadingId) ? prev : [...prev, targetHeadingId])
   427→    setSelectedId(draggedId)
   428→    toast.success('Item reparented', {
   429→      description: `${draggedItem?.code} moved under ${targetHeadingId}`,
   430→    })
   431→  }

   790→/**
   791→ * Draggable + Droppable wrapper for BOQ rows.
   792→ * - All rows are draggable (useDraggable) so they can be moved.
   793→ * - Heading rows are also droppable (useDroppable) so items can be reparented under them.
   794→ * - Shows a drag handle (GripVertical) on hover.
   795→ * - Highlights heading rows when another item is dragged over them.
   796→ */
   797→function BoqDndRow({
   798→  item,
   799→  isHeading,
   800→  isDragged,
   801→  isDragOver,
   802→  children,
   803→}: {
   804→  item: BoqItem
   805→  isHeading: boolean
   806→  isDragged: boolean
   807→  isDragOver: boolean
   808→  children: React.ReactNode
   809→}) {
   810→  const { attributes, listeners, setNodeRef, isDragging } = useDraggable({
   811→    id: item.id,
   812→    disabled: false,
   813→  })
   814→
   815→  const droppable = useDroppable({
   816→    id: item.id,
   817→    disabled: !isHeading,
   818→  })
   819→
   820→  return (
   821→    <div
   822→      ref={(node) => {
   823→        setNodeRef(node)
   824→        if (isHeading) droppable.setNodeRef(node)
   825→      }}
   826→      {...attributes}
   827→      {...listeners}
   828→      className={cn(
   829→        'relative group/dnd',
   830→        isDragging && 'opacity-40',
   831→        isHeading && isDragOver && !isDragged && 'ring-2 ring-primary ring-inset bg-primary/5',
   832→        droppable.isOver && isHeading && 'bg-primary/10',
   833→      )}
   834→    >
   835→      {children}
   836→      {/* Drag handle — appears on hover */}
   837→      <div className={cn(
   838→        'absolute left-0 top-0 bottom-0 w-5 flex items-center justify-center cursor-grab opacity-0 group-hover/dnd:opacity-100 transition-opacity',
   839→        isDragging && 'cursor-grabbing'
   840→      )}>
   841→        <GripVertical className="w-3 h-3 text-muted-foreground/60" />
   842→      </div>
   843→    </div>
   844→  )
   845→}
